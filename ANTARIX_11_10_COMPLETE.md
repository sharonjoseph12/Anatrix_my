# 6. CORE ALGORITHMS (11/10)

## Algorithm 1: Skill Proof Score Calculation

```typescript
// Runs weekly, updates user_skills table

async function calculateSkillProofScore(
  userId: string,
  skillId: string
): Promise<number> {
  
  // Get all relevant data
  const sessions = await db.query(
    `SELECT * FROM sessions 
     WHERE user_id = $1 
     AND category IN (SELECT category FROM skill_categories WHERE skill_id = $2)
     AND started_at > NOW() - INTERVAL '90 days'`,
    [userId, skillId]
  );
  
  const githubActivity = await db.query(
    `SELECT * FROM github_activity
     WHERE user_id = $1
     AND primary_language IN (SELECT language FROM skill_languages WHERE skill_id = $2)
     AND committed_at > NOW() - INTERVAL '90 days'`,
    [userId, skillId]
  );
  
  const projects = await db.query(
    `SELECT * FROM projects
     WHERE user_id = $1
     AND category IN (SELECT category FROM skill_projects WHERE skill_id = $2)
     AND completed_at > NOW() - INTERVAL '90 days'`,
    [userId, skillId]
  );
  
  // Calculate components (each 0-100)
  
  // 1. Hours Component (25% weight)
  const hoursLogged = sessions.reduce((sum, s) => sum + s.durationMinutes, 0) / 60;
  const skill = await db.query('SELECT avg_hours_to_proficiency FROM skills WHERE id = $1', [skillId]);
  const hoursScore = Math.min((hoursLogged / skill.avg_hours_to_proficiency) * 100, 100);
  
  // 2. Projects Component (35% weight)
  const projectsCompleted = projects.filter(p => p.completionStatus === 'completed').length;
  const totalProjects = projects.length;
  const projectCompletionRate = totalProjects > 0 ? projectsCompleted / totalProjects : 0;
  const projectsScore = projectCompletionRate * 100;
  
  // 3. Quality Component (25% weight)
  const avgFocusQuality = sessions.length > 0
    ? sessions.reduce((sum, s) => {
        if (s.focusLevel === 'high') return sum + 100;
        if (s.focusLevel === 'medium') return sum + 60;
        return sum + 30;
      }, 0) / sessions.length
    : 0;
  const qualityScore = avgFocusQuality;
  
  // 4. Consistency Component (15% weight)
  const daysActive = new Set(sessions.map(s => new Date(s.startedAt).toDateString())).size;
  const consistencyScore = Math.min((daysActive / 30) * 100, 100); // Days active in last 30 days
  
  // Weighted average
  const skillProofScore = Math.round(
    (hoursScore * 0.25) +
    (projectsScore * 0.35) +
    (qualityScore * 0.25) +
    (consistencyScore * 0.15)
  );
  
  // Determine proficiency level
  let proficiencyLevel = 'beginner';
  if (skillProofScore >= 75) proficiencyLevel = 'advanced';
  else if (skillProofScore >= 50) proficiencyLevel = 'intermediate';
  
  // Store in database
  await db.query(
    `UPDATE user_skills
     SET skill_proof_score = $1, proficiency_level = $2, updated_at = NOW()
     WHERE user_id = $3 AND skill_id = $4`,
    [skillProofScore, proficiencyLevel, userId, skillId]
  );
  
  return skillProofScore;
}
```

## Algorithm 2: Overall Profile Score

```typescript
// Runs daily, updates candidate_profiles table

async function updateCandidateProfile(userId: string) {
  
  const userSkills = await db.query(
    `SELECT * FROM user_skills WHERE user_id = $1 ORDER BY skill_proof_score DESC`,
    [userId]
  );
  
  if (userSkills.length === 0) {
    // No skills yet, score is 0
    await db.query(
      `UPDATE candidate_profiles SET overall_skill_proof_score = 0 WHERE user_id = $1`,
      [userId]
    );
    return;
  }
  
  // Overall score = weighted average of top 3 skills + bonus for breadth
  const topSkills = userSkills.slice(0, 3);
  const topSkillsScore = topSkills.reduce((sum, s) => sum + s.skill_proof_score, 0) / topSkills.length;
  
  // Breadth bonus (extra points for having 5+ skills)
  const breadthBonus = Math.min((userSkills.length - 1) * 2, 10); // Max +10 bonus
  
  // Specialize bonus (extra points if any skill > 85)
  const specialization = userSkills.find(s => s.skill_proof_score > 85);
  const specializationBonus = specialization ? 5 : 0;
  
  const overallScore = Math.min(
    topSkillsScore + breadthBonus + specializationBonus,
    100
  );
  
  // Get specializations
  const specializationScores = {};
  for (const skill of userSkills.slice(0, 5)) {
    const skillName = await db.query('SELECT name FROM skills WHERE id = $1', [skill.skillId]);
    specializationScores[skillName] = skill.skill_proof_score;
  }
  
  // Determine placement readiness
  const placementReady = overallScore >= 80 && userSkills[0].hoursLogged >= 200;
  
  // Update candidate profile
  await db.query(
    `UPDATE candidate_profiles
     SET
       overall_skill_proof_score = $1,
       specialization_scores = $2,
       placement_ready = $3,
       updated_at = NOW()
     WHERE user_id = $4`,
```

```typescript
// Runs daily, generates insights for colleges

async function generateInstitutionInsights(institutionId: string) {
  
  const members = await db.query(
    `SELECT * FROM institution_members WHERE institution_id = $1`,
    [institutionId]
  );
  
  // 1. Placement Readiness Segmentation
  const placementReadyCount = (await db.query(
    `SELECT COUNT(*) FROM candidate_profiles
     WHERE user_id IN (${members.map(m => `'${m.user_id}'`).join(',')})
     AND placement_ready = true`
  ))[0].count;
  
  // 2. Skill Gap Analysis
  const allSkills = await db.query(
    `SELECT skill_id, COUNT(*) as count, AVG(skill_proof_score) as avg_score
     FROM user_skills
     WHERE user_id IN (${members.map(m => `'${m.user_id}'`).join(',')})
     GROUP BY skill_id
     ORDER BY avg_score DESC`
  );
  
  // Identify gaps: What skills are industry-demanded but underrepresented?
  const industryDemand = {
    'ML': 9, 'Python': 10, 'DevOps': 8, 'Web Dev': 7, 'Cloud': 8
  };
  
  const gaps = [];
  for (const [skill, demand] of Object.entries(industryDemand)) {
    const studentCount = allSkills.find(s => s.name === skill)?.count || 0;
    if (studentCount < members.length * 0.1) { // Less than 10% have this skill
      gaps.push({
        skill,
        demand,
        currentStudents: studentCount,
        gap: Math.round(members.length * 0.2) - studentCount
      });
    }
  }
  
  // 3. Alumni Success Tracking
  const class2023 = await db.query(
    `SELECT cp.user_id, cp.overall_skill_proof_score
     FROM candidate_profiles cp
     JOIN institution_members im ON cp.user_id = im.user_id
     WHERE im.institution_id = $1 AND im.batch_year = 2023`,
    [institutionId]
  );
  
  const placed2023 = (await db.query(
    `SELECT COUNT(*) FROM job_matches
     WHERE candidate_id IN (${class2023.map(c => `'${c.user_id}'`).join(',')})
     AND hired = true`
  ))[0].count;
  
  const placementRate = (placed2023 / class2023.length * 100).toFixed(1);
  
  // Store insights
  const insights = {
    placementReady: placementReadyCount,
    skillGaps: gaps,
    class2023PlacementRate: placementRate,
    recommendedCurriculumFocus: gaps[0]?.skill
  };
  
  await db.query(
    `UPDATE institutions SET 
     placement_rate = $1,
     updated_at = NOW()
     WHERE id = $2`,
    [placementRate / 100, institutionId]
  );
  
  return insights;
}
```

```typescript
// Runs on-demand when company searches

async function findMatchingCandidates(
  companyId: string,
  filters: {
    skills: string[],
    minScore: number,
    batchYears: number[],
    locations: string[]
  }
) {
  
  // Build candidate query
  let query = `
    SELECT cp.*, u.display_name, u.email, im.institution_id
    FROM candidate_profiles cp
    JOIN users u ON cp.user_id = u.id
    LEFT JOIN institution_members im ON cp.user_id = im.user_id
    WHERE cp.is_open_to_opportunities = true
  `;
  
  const params = [];
  
  // Score filter
  query += ` AND cp.overall_skill_proof_score >= $${++params.length}`;
  params.push(filters.minScore);
  
  // Batch year filter
  if (filters.batchYears.length > 0) {
    query += ` AND im.batch_year IN (${filters.batchYears.join(',')})`;
  }
  
  // Location filter
  if (filters.locations.length > 0) {
    query += ` AND u.location IN (${filters.locations.map(() => `$${++params.length}`).join(',')})`;
    params.push(...filters.locations);
  }
  
  const candidates = await db.query(query, params);
  
  // Calculate match score for each candidate
  const matchedCandidates = candidates.map(candidate => {
    let matchScore = 0;
    
    // Skill match
    const skillMatches = filters.skills.filter(
      skill => candidate.specialization_scores[skill] !== undefined
    ).length;
    const skillMatchScore = (skillMatches / filters.skills.length) * 100;
    matchScore += skillMatchScore * 0.5;
    
    // Experience match
    const experienceMatch = candidate.overall_skill_proof_score;
    matchScore += experienceMatch * 0.3;
    
    // Location match (bonus if same city)
    const locationMatch = candidate.location === filters.locations[0] ? 20 : 0;
    matchScore += locationMatch * 0.2;
    
    return {
      ...candidate,
      matchScore: Math.round(matchScore),
      recommendedRole: filters.skills.find(s => 
        candidate.specialization_scores[s] > 80
      ) || 'General'
    };
  });
  
  // Sort by match score
  return matchedCandidates.sort((a, b) => b.matchScore - a.matchScore);
}
```
